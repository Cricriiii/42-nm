/**
 * Sections as definied by man 5 elf.
 * Note: man contains a mistake for .gnu.version_r.
 * Proper type is indeed SHT_GNU_verneed.
 */


void executable_header_interpret(const unsigned char e_ident[16]) {
    printf("Format: %c%c%c%c\n", e_ident[EI_MAG0], e_ident[EI_MAG1],
           e_ident[EI_MAG2], e_ident[EI_MAG3]);

    switch (e_ident[EI_CLASS]) {
        case ELFCLASSNONE:
            printf("Class: %s\n", "This class is invalid.");
            break;
        case ELFCLASS32:
            printf("Class: %s\n", "This defines the 32-bit architecture.");
            break;
        case ELFCLASS64:
            printf("Class: %s\n", "This defines the 64-bit architecture.");
            break;
    }
    switch (e_ident[EI_DATA]) {
        case ELFDATANONE:
            printf("Data: %s\n", "Unknown data format.");
            break;
        case ELFDATA2LSB:
            printf("Data: %s\n", "Two's complement, little-endian.");
            break;
        case ELFDATA2MSB:
            printf("Data: %s\n", "Two's complement, big-endian.");
            break;
    }

    switch (e_ident[EI_VERSION]) {
        case EV_NONE:
            printf("Version: %s\n", "Invalid version.");
            break;
        case EV_CURRENT:
            printf("Version: %s\n", "Current version.");
            break;
    }

    switch (e_ident[EI_OSABI]) {
        case ELFOSABI_NONE:  // same as case ELFOSABI_SYSV:
            printf("OS_ABI: %s\n", "Same as ELFOSABI_SYSV");
            break;
        case ELFOSABI_HPUX:
            printf("OS_ABI: %s\n", "HP-UX ABI");
            break;
        case ELFOSABI_NETBSD:
            printf("OS_ABI: %s\n", "NetBSD ABI");
            break;
        case ELFOSABI_LINUX:
            printf("OS_ABI: %s\n", "Linux ABI");
            break;
        case ELFOSABI_SOLARIS:
            printf("OS_ABI: %s\n", "Solaris ABI");
            break;
        case ELFOSABI_IRIX:
            printf("OS_ABI: %s\n", "IRIX ABI");
            break;
        case ELFOSABI_FREEBSD:
            printf("OS_ABI: %s\n", "FreeBSD ABI");
            break;
        case ELFOSABI_TRU64:
            printf("OS_ABI: %s\n", "TRU64 UNIX ABI");
            break;
        case ELFOSABI_ARM:
            printf("OS_ABI: %s\n", "ARM architecture ABI");
            break;
        case ELFOSABI_STANDALONE:
            printf("OS_ABI: %s\n", "Stand-alone (embedded) ABI");
            break;
    }

    printf("ABI_VERSION: %d\n", e_ident[EI_ABIVERSION]);
    printf("PAD: %d\n", e_ident[EI_PAD]);
    printf("N_INDENT: %d\n", e_ident[EI_NIDENT]);
}

    printf("\n***Section ELF header start***\n");

    executable_header_interpret(e->e_ident);

    printf("shnum: 0x%.4x\n", e->e_shnum);

    printf("e_shstrndx: 0x%.4x\n", e->e_shstrndx);
}

void phdr_interpret(const Elf64_Phdr* p) {
    static int i;

    printf("\n***Section program header #%d start***\n", i++);

    printf("Type: ");
    switch (p->p_type) {
        case PT_NULL:
            printf("PT_NULL\n");
            break;

        case PT_LOAD:
            printf("PT_LOAD\n");
            break;

        case PT_DYNAMIC:
            printf("PT_DYNAMIC\n");
            break;

        case PT_INTERP:
            printf("PT_INTERP\n");
            break;

        case PT_NOTE:
            printf("PT_NOTE\n");
            break;

        case PT_SHLIB:
            printf("PT_SHLIB\n");
            break;

        case PT_PHDR:
            printf("PT_PHDR\n");
            break;

        case PT_GNU_EH_FRAME:
            printf("PT_GNU_EH_FRAME\n");
            break;

        case PT_GNU_STACK:
            printf("PT_GNU_STACK\n");
            break;

        case PT_GNU_RELRO:
            printf("PT_GNU_RELRO\n");
            break;

        case PT_GNU_PROPERTY:
            printf("PT_GNU_PROPERTY\n");
            break;

        default:
            if (p->p_type >= PT_LOPROC && p->p_type <= PT_HIPROC)
                printf("Between PT_LOPROC and PT_HIPROC\n");
            else
                printf("Unknown (0x%x)\n", p->p_type);
            break;
    }

    printf("Offset: 0x%.4lx\n", p->p_offset);
    printf("Vaddr: 0x%.4lx\n", p->p_vaddr);
    printf("Paddr: 0x%.4lx\n", p->p_paddr);

    printf("filesz: 0x%.4lx\n", p->p_filesz);
    printf("memsz: 0x%.4lx\n", p->p_memsz);

    printf("Flags: %s%s%s\n", p->p_flags & PF_R ? "R" : "-",
           p->p_flags & PF_W ? "W" : "-", p->p_flags & PF_X ? "X" : "-");
    printf("Align: %lu\n", p->p_align);

    printf("***Section end***\n\n");
}

void shdr_interpret(const Elf64_Shdr* s, const void* e,
                    Elf64_Off shstrtab_start) {
    static int i;

    printf("\n***Section Section header #%d start***\n", i++);
    
    const char* section_name = (const char*)((e + shstrtab_start) + s->sh_name);
    printf("Name: %s\n", section_name);

    printf("Type: ");
    switch (s->sh_type) {
        case SHT_NULL:
            printf("SHT_NULL\n");
            break;

        case SHT_PROGBITS:
            printf("SHT_PROGBITS\n");
            break;

        case SHT_SYMTAB:
            printf("SHT_SYMTAB\n");

            /* Symbol table */
   
            printf("\n---Symbols start---\n");

            Elf64_Xword size = s->sh_size;
            ptrdiff_t n_elem = (ptrdiff_t)(size / sizeof(Elf64_Sym));
            Elf64_Sym* symtab = (void*)e + s->sh_offset;
            for (Elf64_Sym* cursym = symtab; cursym - symtab < n_elem; ++cursym) {
                printf("%s\n", (const char*)(strtab_addr + cursym->st_name));
                continue;
            }

            printf("***Section end***\n\n");  
            break;

        case SHT_STRTAB:
            if (!strcmp(section_name, ".strtab")) {
                strtab_addr = (void*)e + s->sh_offset;
            } else if (!strcmp(section_name, ".dynstr")) {
                dyn_strtab_addr = (void*)e + s->sh_offset;
            }

            printf("SHT_STRTAB\n");
            break;

        case SHT_RELA:
            printf("SHT_RELA\n");
            break;

        case SHT_HASH:
            printf("SHT_HASH\n");
            break;

        case SHT_DYNAMIC:
            printf("SHT_DYNAMIC\n");
            break;

        case SHT_NOTE:
            printf("SHT_NOTE\n");
            break;

        case SHT_NOBITS:
            printf("SHT_NOBITS\n");
            break;

        case SHT_REL:
            printf("SHT_REL\n");
            break;

        case SHT_SHLIB:
            printf("SHT_SHLIB\n");
            break;

        case SHT_DYNSYM:
            printf("SHT_DYNSYM\n");
            break;

        default:
            if (s->sh_type >= SHT_LOPROC && s->sh_type <= SHT_HIPROC) {
                printf("Between SHT_LOPROC and SHT_HIPROC\n");
                break;
            } else if (s->sh_type >= SHT_LOUSER && s->sh_type <= SHT_HIUSER) {
                printf("Between SHT_LOUSER and SHT_HIUSER\n");
                break;
            }
    }

    printf("Flags: %s%s%s%s\n", s->sh_flags & SHF_WRITE ? "W" : "-",
           s->sh_flags & SHF_ALLOC ? "A" : "-",
           s->sh_flags & SHF_EXECINSTR ? "X" : "-",
           s->sh_flags & SHF_MASKPROC ? "M" : "-");

    printf("Addr: 0x%.4lx\n", s->sh_addr);
    printf("Offset: 0x%.4lx\n", s->sh_offset);
    printf("Size: 0x%.4lx\n", s->sh_size);

    printf("Link: 0x%.4x\n", s->sh_link);
    printf("Info: 0x%.4x\n", s->sh_info);

    printf("Addralign: 0x%.4lx\n", s->sh_addralign);

    printf("Entsize: 0x%.4lx\n", s->sh_entsize);
}
